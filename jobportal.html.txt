            },
            {
                id: 8,
                title: "Product Design Lead",
                company: "Notion",
                logoIcon: "fas fa-cube",
                location: "San Francisco, CA",
                type: "Full-time",
                remote: false,
                tags: ["Design Systems", "Leadership", "UI"],
                salary: "$170k - $210k",
                category: "design"
            }
        ];

        // ---------- DOM refs ----------
        const jobGrid = document.getElementById('jobGrid');
        const jobCountSpan = document.getElementById('jobCount');
        const searchInput = document.getElementById('searchInput');
        const searchBtn = document.getElementById('searchBtn');
        const filterContainer = document.getElementById('filterContainer');
        const toast = document.getElementById('toastMessage');

        // ---------- state ----------
        let activeFilter = 'all';    // 'all', 'remote', 'full-time', 'design', 'engineering'
        let searchQuery = '';

        // ---------- helper: show toast ----------
        function showToast(message) {
            toast.textContent = message;
            toast.classList.add('show');
            setTimeout(() => {
                toast.classList.remove('show');
            }, 2200);
        }

        // ---------- render jobs based on filters & search ----------
        function renderJobs() {
            // Start with all jobs
            let filtered = [...jobs];

            // Apply filter
            if (activeFilter === 'remote') {
                filtered = filtered.filter(job => job.remote === true);
            } else if (activeFilter === 'full-time') {
                filtered = filtered.filter(job => job.type.toLowerCase() === 'full-time');
            } else if (activeFilter === 'design') {
                filtered = filtered.filter(job => job.category === 'design');
            } else if (activeFilter === 'engineering') {
                filtered = filtered.filter(job => job.category === 'engineering');
            }
            // 'all' -> no filter

            // Apply search query
            if (searchQuery.trim() !== '') {
                const query = searchQuery.trim().toLowerCase();
                filtered = filtered.filter(job => 
                    job.title.toLowerCase().includes(query) ||
                    job.company.toLowerCase().includes(query) ||
                    job.tags.some(tag => tag.toLowerCase().includes(query)) ||
                    job.location.toLowerCase().includes(query)
                );
            }

            // Update job count
            jobCountSpan.textContent = `${filtered.length} job${filtered.length !== 1 ? 's' : ''}`;

            // If no jobs, show empty state
            if (filtered.length === 0) {
                jobGrid.innerHTML = `
                    <div class="no-results">
                        <i class="fas fa-search-minus"></i>
                        <h3 style="margin-bottom: 0.5rem; font-weight: 600;">No matching jobs</h3>
                        <p style="color: #64748b;">Try adjusting your search or filters.</p>
                    </div>
                `;
                return;
            }

            // Generate cards
            const cardsHTML = filtered.map(job => {
                // determine icon for company logo — fallback to building icon if not font awesome
                const logoIcon = job.logoIcon || 'fas fa-building';
                return `
                    <div class="job-card" data-job-id="${job.id}">
                        <div class="card-header">
                            <div class="company-logo">
                                <i class="${logoIcon}"></i>
                            </div>
                            <div>
                                <div class="job-title">${job.title}</div>
                                <div class="company-name">${job.company}</div>
                            </div>
                        </div>
                        <div class="job-meta">
                            <span><i class="fas fa-map-marker-alt"></i> ${job.location}</span>
                            <span><i class="fas fa-clock"></i> ${job.type}</span>
                            ${job.remote ? '<span><i class="fas fa-wifi"></i> Remote</span>' : ''}
                        </div>
                        <div class="job-tags">
                            ${job.tags.map(tag => `<span class="tag">${tag}</span>`).join('')}
                        </div>
                        <div class="card-footer">
                            <div class="salary">${job.salary} <span>/ year</span></div>
                            <button class="apply-btn" data-apply-id="${job.id}">
                                Apply <i class="fas fa-arrow-right"></i>
                            </button>
                        </div>
                    </div>
                `;
            }).join('');

            jobGrid.innerHTML = cardsHTML;
        }

        // ---------- update active filter UI ----------
        function setActiveFilter(filterValue) {
            activeFilter = filterValue;
            // update chips
            document.querySelectorAll('.filter-chip').forEach(chip => {
                const chipFilter = chip.getAttribute('data-filter');
                if (chipFilter === filterValue) {
                    chip.classList.add('active');
                } else {
                    chip.classList.remove('active');
                }
            });
            renderJobs();
        }

        // ---------- event listeners ----------

        // Search button click
        searchBtn.addEventListener('click', () => {
            searchQuery = searchInput.value;
            renderJobs();
            if (searchQuery.trim() !== '') {
                showToast(`Searching for "${searchQuery}"`);
            }
        });

        // Enter key in search input
        searchInput.addEventListener('keyup', (e) => {
            if (e.key === 'Enter') {
                searchQuery = searchInput.value;
                renderJobs();
                if (searchQuery.trim() !== '') {
                    showToast(`Searching for "${searchQuery}"`);
                }
            }
        });

        // Live search (optional: mild delay but okay)
        searchInput.addEventListener('input', (e) => {
            // Optional: real-time filter, but only if you want instant feedback
            // I'll keep it snappy: update searchQuery and re-render immediately (no debounce for simplicity)
            searchQuery = e.target.value;
            renderJobs();
        });

        // Filter chips
        filterContainer.addEventListener('click', (e) => {
            const chip = e.target.closest('.filter-chip');
            if (!chip) return;
            const filter = chip.getAttribute('data-filter');
            setActiveFilter(filter);
            // show toast for filter feedback
            const filterNames = {
                all: 'All jobs', remote: 'Remote', 'full-time': 'Full-time',
                design: 'Design', engineering: 'Engineering'
            };
            showToast(`Filter: ${filterNames[filter] || filter}`);
        });

        // Apply button click (delegation)
        jobGrid.addEventListener('click', (e) => {
            // if click on apply button (or inside)
            const applyBtn = e.target.closest('.apply-btn');
            if (applyBtn) {
                e.stopPropagation(); // prevent card click
                const jobId = applyBtn.getAttribute('data-apply-id');
                const job = jobs.find(j => j.id == jobId);
                if (job) {
                    showToast(`📄 Application started for ${job.title} at ${job.company}`);
                } else {
                    showToast('Application started');
                }
                return;
            }

            // Optional: clicking on card (not the apply button) – show a subtle toast
            const card = e.target.closest('.job-card');
            if (card) {
                const jobId = card.getAttribute('data-job-id');
                const job = jobs.find(j => j.id == jobId);
                if (job) {
                    showToast(`🔍 ${job.title} at ${job.company} — details preview`);
                }
            }
        });

        // Sign in / Post job buttons: dummy toasts
        document.getElementById('loginBtn').addEventListener('click', () => {
            showToast('👤 Sign in flow (demo)');
        });
        document.getElementById('postJobBtn').addEventListener('click', () => {
            showToast('📢 Post a job (demo)');
        });

        // ---------- initial render ----------
        renderJobs();

        // (Optional) prefill search placeholder with something subtle
        // nothing else
    })();
